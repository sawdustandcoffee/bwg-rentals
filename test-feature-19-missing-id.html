Function: property_card() (lines 514-541)

public function property_card( $atts ) { $this->enqueue_assets(); $atts = shortcode_atts( array( 'id' => 0, 'show_image' => 'true', 'show_specs' => 'true', 'link' => 'true', ), $atts, 'bwg_property_card' ); // LINE 528-530: Error handling for missing ID if ( empty( $atts['id'] ) ) { return $this->render_error( __( 'Property ID is required.', 'bwg-rentals' ) ); } $property = $this->api->get_property( $atts['id'] ); if ( is_wp_error( $property ) ) { return $this->render_error( $property->get_error_message() ); } ob_start(); include $this->get_template( 'property-card.php' ); return ob_get_clean(); }
